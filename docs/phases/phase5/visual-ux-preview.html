it('断流后按 seq 续传，不重复渲染', async () => {
  const events = [
    { seq: 41, type: 'content', data: '{"text":"A"}' },
    { seq: 42, type: 'content', data: '{"text":"B"}' }
  ]
  const seen: string[] = []
  await parseSSE(streamOf(events), (t, d) => {
    if (t === 'content') seen.push(JSON.parse(d).text)
  })
  expect(seen).toEqual(['A', 'B'])   // 顺序与去重
})
关键路径 E2E Playwright
test('发送 → 流式 → 审批 → 完成', async ({ page }) => {
  await page.goto('/chat')
